function finSum(){
  const p=[];
  if(S.res){let tp=0;S.res.secs.forEach(r=>{const s=SECS.find(x=>x.id===r.id);tp+=r.ok*s.pts});p.push(`✍️ Schriftlich: <b>${fp(tp)}/150</b>`)}
  const d=(S.mund&&S.mund.done)||[];
  if(S.mode!=="S"&&d.length)p.push(`🗣 Mündlich: <b>${mTotal()}/75</b> (${d.length}/3 Teil)`);
  return p;
}
function finStart(){
  if(!S)return;
  try{window.AISprech&&window.AISprech.unmount()}catch(e){}
  const ph=S.phase,mode=S.mode||"S",now=Date.now();
  if(mode!=="M"&&(ph==="L"||ph==="H"||ph==="mid")){
    if(!S.end.L)S.end.L=now;if(ph==="H")S.end.H=now;
    S.res={id:S.id,ts:now,mode:mode,secs:SECS.map(gradeSec),usedL:Math.max(1,Math.round((S.end.L-S.t0.L)/60000)),
      usedH:S.t0.H?Math.max(1,Math.round(((S.end.H||now)-S.t0.H)/60000)):0,auto:S.auto||{},early:ph};
    S.sent=false;
  }
  const d=(S.mund&&S.mund.done)||[];
  S.fin={from:ph,needS:!!(S.res&&!S.sent),needM:mode!=="S"&&d.length>0&&!S.mund.sent,sum:finSum()};
  S.phase="fin";save();render();window.scrollTo(0,0);
}
function viewFin(){
  const f=S.fin||{sum:[]};
  app.innerHTML=`<div id="bar"><span>⏹ Imtihon yakunlanmoqda</span></div><main><div class="card"><h2>Imtihon yakunlandi</h2>
  ${f.sum.length?`<ul class="rules">${f.sum.map(x=>`<li>${x}</li>`).join("")}</ul>`:`<p style="color:var(--muted)">Hech qaysi qism tugallanmagan — hisoblanadigan natija yo'q.</p>`}
  <div id="snd" style="text-align:center;font-weight:600;color:var(--muted)"></div><p class="msg" id="msg"></p></div></main>`;
  finSend();
}
async function finSend(){
  const f=S.fin;if(!f)return;
  const st=h=>{const e=$("#snd");if(e)e.innerHTML=h};
  st("⏳ Natija ustozga yuborilmoqda…");
  const post=b=>fetch(X.fn+"/submit",{method:"POST",headers:{"content-type":"application/json"},body:JSON.stringify(b)}).then(r=>r.json());
  try{
    if(f.needS&&!S.sent){const r=await post({initData:X.initData,res:S.res});if(!(r&&r.ok))throw 0;S.sent=true;save()}
    if(f.needM&&!S.mund.sent){
      const teile=S.mund.done.map(x=>({teil:x.teil,title:String(x.title||"").slice(0,80),s:x.s,tot:x.tot}));
      const r=await post({initData:X.initData,kind:"mund",mode:S.mode,early:true,mund:{id:S.id,teile},res:S.mode==="G"?S.res:null});
      if(!(r&&r.ok))throw 0;S.mund.sent=true;save()}
  }catch(e){
    st('⚠️ Natija yuborilmadi. Internetni tekshiring.');
    const m=$("#msg");if(m){m.innerHTML='<button class="go" id="finre">Qayta yuborish</button><button class="go alt" id="finout" style="margin-top:10px">Yubormasdan bosh menyuga</button>';
      $("#finre").onclick=finSend;$("#finout").onclick=finExit}
    return;
  }
  finExit();
}

  S=null;CERT=null;ls.set(KEY,"");SEL=null;render();window.scrollTo(0,0);
}
